import java.util.ArrayList;
import java.util.Scanner;

// Module 4: Class, object and constructor
class Patient {
    int id;
    String name;
    int age;
    String symptoms;
    String priority;

    Patient(int id, String name, int age,
            String symptoms, String priority) {
        this.id = id;
        this.name = name;
        this.age = age;
        this.symptoms = symptoms;
        this.priority = priority;
    }

    void display() {
        System.out.println("Patient ID: " + id);
        System.out.println("Name: " + name);
        System.out.println("Age: " + age);
        System.out.println("Symptoms: " + symptoms);
        System.out.println("Priority: " + priority);
        System.out.println("----------------------");
    }
}

public class HospitalTriageSystem {

    // Module 6: ArrayList collection
    static ArrayList<Patient> patients = new ArrayList<>();

    // Module 1: Variables, data types and Scanner
    static Scanner sc = new Scanner(System.in);

    // Module 3: User-defined method
    static void registerPatient() {
        System.out.print("Enter Patient ID: ");
        int id = sc.nextInt();
        sc.nextLine();

        System.out.print("Enter Patient Name: ");
        String name = sc.nextLine();

        System.out.print("Enter Patient Age: ");
        int age = sc.nextInt();
        sc.nextLine();

        System.out.print("Enter Symptoms: ");
        String symptoms = sc.nextLine();

        System.out.println("Select Priority:");
        System.out.println("1. Critical");
        System.out.println("2. Urgent");
        System.out.println("3. Normal");
        System.out.print("Enter choice: ");
        int choice = sc.nextInt();
        sc.nextLine();

        // Module 2: if-else selection
        String priority;

        if (choice == 1) {
            priority = "Critical";
        } else if (choice == 2) {
            priority = "Urgent";
        } else if (choice == 3) {
            priority = "Normal";
        } else {
            System.out.println("Invalid priority!");
            return;
        }

        Patient p = new Patient(
            id, name, age, symptoms, priority
        );

        patients.add(p);

        System.out.println("Patient registered successfully!");
    }

    static void displayPatients() {
        if (patients.isEmpty()) {
            System.out.println("No patients registered.");
            return;
        }

        for (Patient p : patients) {
            p.display();
        }
    }

    // Module 5: String comparison and searching
    static void searchPatient() {
        sc.nextLine();

        System.out.print("Enter patient name to search: ");
        String searchName = sc.nextLine();

        boolean found = false;

        for (Patient p : patients) {
            if (p.name.equalsIgnoreCase(searchName)) {
                p.display();
                found = true;
            }
        }

        if (!found) {
            System.out.println("Patient not found.");
        }
    }

    public static void main(String[] args) {
        int choice;

        // Module 2: do-while loop and switch
        do {
            System.out.println("\n=== HOSPITAL TRIAGE SYSTEM ===");
            System.out.println("1. Register Patient");
            System.out.println("2. Display All Patients");
            System.out.println("3. Search Patient");
            System.out.println("4. Exit");
            System.out.print("Enter choice: ");

            choice = sc.nextInt();

            switch (choice) {
                case 1:
                    registerPatient();
                    break;
                case 2:
                    displayPatients();
                    break;
                case 3:
                    searchPatient();
                    break;
                case 4:
                    System.out.println("Exiting system...");
                    break;
                default:
                    System.out.println("Invalid choice!");
            }

        } while (choice != 4);

        sc.close();
    }
}
